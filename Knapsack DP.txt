
n = int(input("Enter number of items: "))

wt = list(map(int, input("Enter weights of items: ").split()))
p = list(map(int, input("Enter profits of items: ").split()))

W = int(input("Enter knapsack capacity: "))

dp = [[0] * (W + 1) for i in range(n + 1)]

for i in range(1, n + 1):
    for j in range(1, W + 1):
        if wt[i-1] <= j:
            dp[i][j] = max(p[i-1] + dp[i-1][j-wt[i-1]], dp[i-1][j])
        else:
            dp[i][j] = dp[i-1][j]

print("Maximum profit:", dp[n][W])

print("Selected items:", end=" ")
j = W
for i in range(n, 0, -1):
    if dp[i][j] != dp[i-1][j]:
        print(i, end=" ")
        j -= wt[i-1]

print()
print("Time Complexity: O(n * W)")
print("Space Complexity: O(n * W)")


n = int(input("Enter number of queens: "))
r, c = map(int, input("Enter first queen position (row col, 1-based): ").split())

board = [['.' for i in range(n)] for j in range(n)]
board[r-1][c-1] = 'Q'

def safe(row, col):
    for i in range(n):
        for j in range(n):
            if board[i][j] == 'Q':
                if j == col or abs(i-row) == abs(j-col):
                    return False
    return True

def solve(row):
    if row == n:
        return True

    if row == r-1:
        return solve(row+1)

    for col in range(n):
        if safe(row, col):
            board[row][col] = 'Q'

            if solve(row+1):
                return True

            board[row][col] = '.'

    return False

if solve(0):
    print("Solution:")
    for row in board:
        print(*row)
else:
    print("No solution")

print("Worst-case Time Complexity: O(N!)")
print("Space Complexity: O(N^2)")
